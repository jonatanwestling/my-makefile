/**
 * get_cmd_targets()- Creates a list with specified targets.
 *
 * @argc: An array of strings (pointer to a pointer to a char).
 * @argv: The array with strings
 * RETURNS: a pointer to the array that contain the spcified targets
 */
char **get_cmd_targets(int argc, char *argv[]){
	char **cmd = NULL;
	cmd = malloc(sizeof(char*)*(argc - optind));
	if (cmd == NULL)
	{
		perror("Memory allocation failed.");
		exit(EXIT_FAILURE);
	}
	int i = 0;
	int j = optind;
	//loop over and copy the targets
	while (optind < argc)
	{
		if((cmd[i] = strdup(argv[j])) == NULL)
		{
			perror("Memory allocation falied");
			exit(EXIT_FAILURE);
		}
		i++;
		j++;
	}
	return cmd;
}
/**
 * build_targets()- Build the targets either the default one or
 * the specified targets.
 *
 * @m: a pointer to the struct that cantains the parsed_makefile
 * @targetname: a string of the targets name
 * RETURNS: nothing.
 */
void build_targets(makefile *m, const char *targetname, flags *f)
{
	// get the rules of the target
	rule *target_rule = makefile_rule(m, targetname);

	if (target_rule == NULL)
	{
		fprintf(stderr, "Bottom reached!\n");
		return;
	}
	// get the prereqs of the target
	const char **prereq = rule_prereq(target_rule);

	fprintf(stderr, "Target: %s\n", targetname);
	int i = 0;

	while (prereq[i] != NULL)
	{
		fprintf(stderr, "Prereq: %s\n", prereq[i]);
		i++;
	}

	int j = 0;
	// const char *prereq = rule_prereq(target_rule);
	while (prereq[j] != NULL)
	{
		build_targets(m, prereq[j], f);
		j++;
	}
}

bool first_file_is_newer(const char *file1, const char *file2)
{
	struct stat file_info1;
	struct stat file_info2;

	if (stat(file1, &file_info1) != 0)
	{
		perror("stat");
		exit(EXIT_FAILURE);
	}

	if (stat(file2, &file_info2) != 0)
	{
		perror("stat");
		exit(EXIT_FAILURE);
	}

	if (file_info1.st_mtime > file_info2.st_mtime)
	{
		return true;
	}

	return false;
}
